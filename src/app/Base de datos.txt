-- 1. Creación de la base de datos y selección
CREATE DATABASE IF NOT EXISTS josman_db;
USE josman_db;

-- 2. Tabla de Usuarios
CREATE TABLE IF NOT EXISTS usuarios (
    id INT AUTO_INCREMENT PRIMARY KEY,
    nombre VARCHAR(100) NOT NULL,
    correo VARCHAR(100) NOT NULL UNIQUE,
    password VARCHAR(255) NOT NULL,
    edad INT,
    profesion VARCHAR(100),
    fecha_nacimiento DATE,
    domicilio VARCHAR(255),
    colonia VARCHAR(100),
    ciudad VARCHAR(100),
    estado VARCHAR(100),
    creado_en TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    actualizado_en TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

-- 3. NUEVA: Tabla de Productos (Para el catálogo dinámico)
CREATE TABLE IF NOT EXISTS productos (
    id INT AUTO_INCREMENT PRIMARY KEY,
    nombre VARCHAR(255) NOT NULL,
    slug VARCHAR(255) NOT NULL UNIQUE, -- Ejemplo: 'boquilla-sin-arena'
    descripcion TEXT,
    descripcion_larga TEXT,
    precio DECIMAL(10,2) DEFAULT 0.00,
    imagen_url VARCHAR(500), -- Enlace a Supabase/Cloudinary o ruta local
    seccion ENUM('concretos', 'acabados', 'materiales') NOT NULL,
    categoria_slug VARCHAR(100) NOT NULL, -- Ejemplo: 'adhesivos'
    subcategoria_slug VARCHAR(100),
    stock INT DEFAULT 0,
    activo BOOLEAN DEFAULT TRUE,
    creado_en TIMESTAMP DEFAULT CURRENT_TIMESTAMP
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- 4. Tabla de Pedidos (Relacionada con productos por nombre o ID)
CREATE TABLE IF NOT EXISTS pedidos (
    id INT AUTO_INCREMENT PRIMARY KEY,
    usuario_id INT, 
    producto VARCHAR(255), -- Se puede relacionar con productos.nombre
    cantidad INT, 
    total DECIMAL(10,2),
    estado ENUM('pendiente','procesando','completado','cancelado') DEFAULT 'pendiente',
    fecha TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (usuario_id) REFERENCES usuarios(id) ON DELETE CASCADE
) ENGINE=InnoDB;

-- 5. Tabla de Servicios Contratados
CREATE TABLE IF NOT EXISTS servicios_contratados (
    id INT AUTO_INCREMENT PRIMARY KEY,
    usuario_id INT, 
    tipo VARCHAR(100), 
    descripcion TEXT,
    estado ENUM('activo','completado','cancelado') DEFAULT 'activo',
    fecha TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (usuario_id) REFERENCES usuarios(id) ON DELETE CASCADE
) ENGINE=InnoDB;

-- 6. Tabla de Suscripciones
CREATE TABLE IF NOT EXISTS suscripciones (
    id INT AUTO_INCREMENT PRIMARY KEY,
    usuario_id INT,
    plan VARCHAR(100),
    estado ENUM('activa','vencida','cancelada') DEFAULT 'activa',
    fecha_inicio DATE,
    fecha_fin DATE,
    FOREIGN KEY (usuario_id) REFERENCES usuarios(id) ON DELETE CASCADE
) ENGINE=InnoDB;
