-- ============================================================
--  FERCADI / Josman Texturizados — josman_db
--  Esquema completo + seed de datos
--  Ejecutar en phpMyAdmin o MySQL Workbench
-- ============================================================

-- 1. Base de datos
CREATE DATABASE IF NOT EXISTS josman_db
  CHARACTER SET utf8mb4 COLLATE utf8mb4_unicode_ci;
USE josman_db;

-- ============================================================
-- 2. Usuarios
-- ============================================================
CREATE TABLE IF NOT EXISTS usuarios (
    id               INT          AUTO_INCREMENT PRIMARY KEY,
    nombre           VARCHAR(100) NOT NULL,
    correo           VARCHAR(100) NOT NULL UNIQUE,
    password         VARCHAR(255) NOT NULL,
    rol              ENUM('usuario','admin') DEFAULT 'usuario',  -- 'admin' habilita /admin
    edad             INT,
    profesion        VARCHAR(100),
    fecha_nacimiento DATE,
    domicilio        VARCHAR(255),
    colonia          VARCHAR(100),
    ciudad           VARCHAR(100),
    estado           VARCHAR(100),
    creado_en        TIMESTAMP    DEFAULT CURRENT_TIMESTAMP,
    actualizado_en   TIMESTAMP    DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

-- ============================================================
-- 3. Productos del catálogo
-- ============================================================
CREATE TABLE IF NOT EXISTS productos (
    id               INT           AUTO_INCREMENT PRIMARY KEY,
    nombre           VARCHAR(255)  NOT NULL,
    slug             VARCHAR(255)  NOT NULL,
    descripcion      TEXT,
    descripcion2     TEXT,                           -- Segunda descripción (algunos concretos)
    precio           DECIMAL(10,2) DEFAULT 0.00,
    imagen_url       VARCHAR(500),
    seccion          ENUM('concretos','textucos','materiales','ferreteria') NOT NULL,
    categoria_slug   VARCHAR(100)  NOT NULL,         -- Ej: 'adhesivos', 'p085'
    categoria_nombre VARCHAR(255),                   -- Ej: 'Morteros y Afinadores', 'Llaves ajustables'
    stock            INT           DEFAULT 0,
    activo           BOOLEAN       DEFAULT TRUE,
    -- ── Campos públicos nuevos (CSV) ──────────────────────────
    marca            VARCHAR(100),                   -- Ej: 'Truper', 'Expert'
    unidad           VARCHAR(50),                    -- Ej: 'Pieza', 'Set', 'Par'
    -- ── Campos comerciales (solo admin) ──────────────────────
    codigo_interno   VARCHAR(50),                    -- Código interno del proveedor
    ean              VARCHAR(30),                    -- Código de barras (puede venir en notación científica)
    margen           VARCHAR(20),                    -- Margen de mercado: MM00, MM0, MM1…
    caja             SMALLINT UNSIGNED,              -- Unidades por caja
    master           SMALLINT UNSIGNED,              -- Unidades por master
    alta_rotacion    TINYINT(1)    DEFAULT 0,
    precio_minimo    DECIMAL(10,2),                  -- Precio mínimo de venta
    precio_mayoreo_con_iva       DECIMAL(10,2),
    precio_distribuidor_con_iva  DECIMAL(10,2),
    precio_publico_con_iva       DECIMAL(10,2),      -- Igual que `precio` para acceso rápido admin
    precio_mayoreo_sin_iva       DECIMAL(10,2),
    precio_distribuidor_sin_iva  DECIMAL(10,2),
    precio_publico_sin_iva       DECIMAL(10,2),
    precio_medio_mayoreo_sin_iva DECIMAL(10,2),
    precio_medio_mayoreo_con_iva DECIMAL(10,2),
    codigo_sat       VARCHAR(20),                    -- Clasificación SAT
    descripcion_sat  VARCHAR(255),
    peso_kg          DECIMAL(8,3),
    volumen_cm3      DECIMAL(12,3),
    creado_en        TIMESTAMP     DEFAULT CURRENT_TIMESTAMP,
    UNIQUE KEY unique_slug_seccion (slug, seccion),
    INDEX idx_codigo_interno (codigo_interno),
    INDEX idx_marca           (marca),
    INDEX idx_categoria       (seccion, categoria_slug)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- ============================================================
-- 4. Categorías de Materiales (marcas por categoría)
-- ============================================================
CREATE TABLE IF NOT EXISTS materiales_categorias (
    id          INT          AUTO_INCREMENT PRIMARY KEY,
    slug        VARCHAR(100) NOT NULL UNIQUE,
    nombre      VARCHAR(255) NOT NULL,
    descripcion TEXT,
    marcas      JSON,                                -- Array: [{nombre, logo}]
    activo      BOOLEAN      DEFAULT TRUE,
    creado_en   TIMESTAMP    DEFAULT CURRENT_TIMESTAMP
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- ============================================================
-- 5. Pedidos
-- ============================================================
CREATE TABLE IF NOT EXISTS pedidos (
    id              INT           AUTO_INCREMENT PRIMARY KEY,
    usuario_id      INT,
    producto        VARCHAR(255),
    opciones        VARCHAR(255),                    -- Presentación/tamaño elegido (puede ser NULL)
    cantidad        INT,
    precio_unitario DECIMAL(10,2),                  -- Precio por unidad al momento de la compra
    total           DECIMAL(10,2),
    estado          ENUM('pendiente','procesando','completado','cancelado') DEFAULT 'pendiente',
    fecha           TIMESTAMP     DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (usuario_id) REFERENCES usuarios(id) ON DELETE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- ============================================================
-- 6. Órdenes (agrupa los ítems de un mismo carrito)
-- ============================================================
CREATE TABLE IF NOT EXISTS ordenes (
    id                INT           AUTO_INCREMENT PRIMARY KEY,
    usuario_id        INT           NOT NULL,
    total             DECIMAL(10,2) NOT NULL DEFAULT 0.00,
    estado            ENUM('pendiente','confirmado','en_preparacion','enviado','entregado','cancelado')
                                    NOT NULL DEFAULT 'pendiente',
    notas             TEXT,
    direccion_entrega TEXT,
    metodo_pago       VARCHAR(50)   NULL,
    created_at        TIMESTAMP     DEFAULT CURRENT_TIMESTAMP,
    updated_at        TIMESTAMP     DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    INDEX idx_usuario (usuario_id),
    INDEX idx_estado  (estado),
    FOREIGN KEY (usuario_id) REFERENCES usuarios(id) ON DELETE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- ============================================================
-- 7. Servicios contratados
-- ============================================================
CREATE TABLE IF NOT EXISTS servicios_contratados (
    id          INT          AUTO_INCREMENT PRIMARY KEY,
    usuario_id  INT,
    tipo        VARCHAR(100),
    descripcion TEXT,
    estado      ENUM('activo','completado','cancelado') DEFAULT 'activo',
    fecha       TIMESTAMP    DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (usuario_id) REFERENCES usuarios(id) ON DELETE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- ============================================================
-- 7. Suscripciones
-- ============================================================
CREATE TABLE IF NOT EXISTS suscripciones (
    id           INT          AUTO_INCREMENT PRIMARY KEY,
    usuario_id   INT,
    plan         VARCHAR(100),
    estado       ENUM('activa','vencida','cancelada') DEFAULT 'activa',
    fecha_inicio DATE,
    fecha_fin    DATE,
    FOREIGN KEY (usuario_id) REFERENCES usuarios(id) ON DELETE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- ============================================================
-- 8. ALTER TABLE — Columnas añadidas en versiones posteriores
--    Si ejecutas este script desde cero, las tablas ya las incluyen.
--    Sólo ejecuta estas líneas si la tabla fue creada antes de los cambios.
-- ============================================================

-- usuarios: columna rol (habilita /admin)
ALTER TABLE usuarios
  ADD COLUMN IF NOT EXISTS rol ENUM('usuario','admin') DEFAULT 'usuario' AFTER password;

-- pedidos: columnas opciones y precio_unitario (historial de compras detallado)
ALTER TABLE pedidos
  ADD COLUMN IF NOT EXISTS opciones        VARCHAR(255)   AFTER producto,
  ADD COLUMN IF NOT EXISTS precio_unitario DECIMAL(10,2)  AFTER cantidad;

-- ordenes: tabla nueva (si la BD fue creada antes del módulo de ventas)
CREATE TABLE IF NOT EXISTS ordenes (
    id                INT           AUTO_INCREMENT PRIMARY KEY,
    usuario_id        INT           NOT NULL,
    total             DECIMAL(10,2) NOT NULL DEFAULT 0.00,
    estado            ENUM('pendiente','confirmado','en_preparacion','enviado','entregado','cancelado')
                                    NOT NULL DEFAULT 'pendiente',
    notas             TEXT,
    direccion_entrega TEXT,
    metodo_pago       VARCHAR(50)   NULL,
    created_at        TIMESTAMP     DEFAULT CURRENT_TIMESTAMP,
    updated_at        TIMESTAMP     DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    INDEX idx_usuario (usuario_id),
    INDEX idx_estado  (estado),
    FOREIGN KEY (usuario_id) REFERENCES usuarios(id) ON DELETE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- pedidos: columna orden_id para vincular ítems a una orden
ALTER TABLE pedidos
  ADD COLUMN IF NOT EXISTS orden_id INT NULL AFTER usuario_id,
  ADD INDEX  IF NOT EXISTS idx_orden_id (orden_id);

-- ordenes: columna metodo_pago (forma de pago elegida en checkout)
ALTER TABLE ordenes
  ADD COLUMN IF NOT EXISTS metodo_pago VARCHAR(50) NULL AFTER direccion_entrega;

-- Para convertir un usuario existente en admin (reemplazar 1 con el id real):
-- UPDATE usuarios SET rol = 'admin' WHERE id = 1;

-- productos: columnas públicas (marca, unidad)
ALTER TABLE productos
  ADD COLUMN IF NOT EXISTS marca   VARCHAR(100) DEFAULT NULL AFTER categoria_nombre,
  ADD COLUMN IF NOT EXISTS unidad  VARCHAR(50)  DEFAULT NULL AFTER marca;

-- productos: columnas comerciales (admin-only)
ALTER TABLE productos
  ADD COLUMN IF NOT EXISTS codigo_interno              VARCHAR(50)      DEFAULT NULL AFTER unidad,
  ADD COLUMN IF NOT EXISTS ean                         VARCHAR(30)      DEFAULT NULL,
  ADD COLUMN IF NOT EXISTS margen                      VARCHAR(20)      DEFAULT NULL,
  ADD COLUMN IF NOT EXISTS caja                        SMALLINT UNSIGNED DEFAULT NULL,
  ADD COLUMN IF NOT EXISTS master                      SMALLINT UNSIGNED DEFAULT NULL,
  ADD COLUMN IF NOT EXISTS alta_rotacion               TINYINT(1)       DEFAULT 0,
  ADD COLUMN IF NOT EXISTS precio_minimo               DECIMAL(10,2)    DEFAULT NULL,
  ADD COLUMN IF NOT EXISTS precio_mayoreo_con_iva      DECIMAL(10,2)    DEFAULT NULL,
  ADD COLUMN IF NOT EXISTS precio_distribuidor_con_iva DECIMAL(10,2)    DEFAULT NULL,
  ADD COLUMN IF NOT EXISTS precio_publico_con_iva      DECIMAL(10,2)    DEFAULT NULL,
  ADD COLUMN IF NOT EXISTS precio_mayoreo_sin_iva      DECIMAL(10,2)    DEFAULT NULL,
  ADD COLUMN IF NOT EXISTS precio_distribuidor_sin_iva DECIMAL(10,2)    DEFAULT NULL,
  ADD COLUMN IF NOT EXISTS precio_publico_sin_iva      DECIMAL(10,2)    DEFAULT NULL,
  ADD COLUMN IF NOT EXISTS precio_medio_mayoreo_sin_iva DECIMAL(10,2)   DEFAULT NULL,
  ADD COLUMN IF NOT EXISTS precio_medio_mayoreo_con_iva DECIMAL(10,2)   DEFAULT NULL,
  ADD COLUMN IF NOT EXISTS codigo_sat                  VARCHAR(20)      DEFAULT NULL,
  ADD COLUMN IF NOT EXISTS descripcion_sat             VARCHAR(255)     DEFAULT NULL,
  ADD COLUMN IF NOT EXISTS peso_kg                     DECIMAL(8,3)     DEFAULT NULL,
  ADD COLUMN IF NOT EXISTS volumen_cm3                 DECIMAL(12,3)    DEFAULT NULL;

-- productos: expandir seccion ENUM para incluir 'ferreteria' (productos del CSV)
ALTER TABLE productos
  MODIFY COLUMN seccion ENUM('concretos','textucos','materiales','ferreteria') NOT NULL;

-- productos: índices para búsquedas del admin
ALTER TABLE productos
  ADD INDEX IF NOT EXISTS idx_codigo_interno (codigo_interno),
  ADD INDEX IF NOT EXISTS idx_marca          (marca),
  ADD INDEX IF NOT EXISTS idx_categoria      (seccion, categoria_slug);
